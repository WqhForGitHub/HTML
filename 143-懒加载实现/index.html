const io = new IntersectionObserver((es) => { es.forEach(e => { if (e.isIntersecting) { e.target.src = e.target.dataset.src; e.target.classList.add('show'); io.unobserve(e.target); } }); }); imgs.forEach(i => io.observe(i));
